// A1: 按类别统计图书数量与平均评分
db.books.aggregate([{ $group: { _id: "$category", count: { $sum: 1 }, avg_rating: { $avg: "$rating" } } }]);

// A2: 各出版社图书数量取前5名
db.books.aggregate([{ $group: { _id: "$publisher", count: { $sum: 1 } } }, { $group: { _id: null, top: { $topN: { n: 5, sortBy: { count: -1 }, output: ["$_id", "$count"] } } } }]);

// A3: 文档数最多的图书类别
db.books.aggregate([{ $sortByCount: "$category" }]);

// A4: 评分最高前3，显示书名与借阅次数
db.books.aggregate([{ $sort: { rating: -1 } }, { $limit: 3 }, { $project: { title: 1, borrow_count: 1, _id: 0 } }]);

// A5: 计算每本书的销售额
db.books.aggregate([{ $project: { title: 1, revenue: { $round: [{ $multiply: ["$price", "$borrow_count"] }, 2] } } }]);

// A6: 按年份分组统计借阅量
db.borrow_logs.aggregate([{ $group: { _id: { $dateToString: { format: "%Y年", date: "$borrow_date" } }, count: { $sum: 1 } } }]);

// A7: 借阅最多的前5名读者
db.borrow_logs.aggregate([{ $group: { _id: "$reader_card", count: { $sum: 1 } } }, { $sort: { count: -1 } }, { $limit: 5 }]);

// A8: 关联读者与借阅流水，学院借阅量排行榜
db.readers.aggregate([{ $lookup: { from: "borrow_logs", localField: "card_no", foreignField: "reader_card", as: "borrows" } }, { $project: { college: 1, b_count: { $size: "$borrows" } } }, { $group: { _id: "$college", total: { $sum: "$b_count" } } }, { $sort: { total: -1 } }]);

// A9: 分类统计结果写入新集合
db.books.aggregate([{ $group: { _id: "$category", count: { $sum: 1 }, avg_rating: { $avg: "$rating" } } }, { $merge: { into: "category_stats" } }]);

from mongodb_connect import get_db

db = get_db()

def category_stats():
    pipeline = [
        {"$group": {"_id": "$category", "total_stock": {"$sum": "$stock"}, "avg_price": {"$avg": "$price"}, "count": {"$sum": 1}}},
        {"$sort": {"total_stock": -1}}
    ]
    for r in db.books.aggregate(pipeline):
        print(f"分类: {r['_id']} | 数量: {r['count']} | 总库存: {r['total_stock']} | 均价: {round(r['avg_price'],2)}")

def top_books():
    pipeline = [{"$sort": {"borrow_count": -1}}, {"$limit": 3}]
    for r in db.books.aggregate(pipeline):
        print(f"书名: {r.get('title')} | 借阅次数: {r.get('borrow_count')}")
